cmake_minimum_required(VERSION 3.24)
project(IsaacGamebot VERSION 2026.11.02 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "The game reader and input backend require Windows.")
endif()
add_library(gamebot_core src/model.cpp src/planner.cpp src/combat.cpp src/combat_progress.cpp src/attack_forecast.cpp src/enemy_intent.cpp src/unknown_threat.cpp src/world_state.cpp src/forward_sim.cpp src/temporal_reachability.cpp src/movement_intent.cpp src/policy_validation.cpp src/projectile_motion.cpp src/safety.cpp src/prediction_monitor.cpp src/lookahead.cpp src/mpc.cpp src/strategy.cpp src/closed_loop.cpp src/performance_monitor.cpp src/session_review.cpp src/discovery.cpp src/benchmark.cpp src/reader.cpp src/semantics.cpp src/control.cpp src/weapons.cpp)
target_include_directories(gamebot_core PUBLIC include third_party)
target_compile_definitions(gamebot_core PUBLIC UNICODE _UNICODE)
target_link_libraries(gamebot_core PUBLIC bcrypt user32 advapi32)
add_executable(IsaacGamebot src/main.cpp src/gui_win32.cpp src/keyboard_win32.cpp)
target_link_libraries(IsaacGamebot PRIVATE gamebot_core gdi32 shell32)
if(MSVC)
  target_link_options(IsaacGamebot PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
else()
  enable_language(RC)
  target_sources(IsaacGamebot PRIVATE app.rc)
endif()
add_executable(GamebotTests tests/core_tests.cpp)
target_link_libraries(GamebotTests PRIVATE gamebot_core)
if(MINGW)
  target_link_options(IsaacGamebot PRIVATE -municode)
endif()
enable_testing()
add_test(NAME GamebotCore COMMAND GamebotTests)
